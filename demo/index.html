import { area, line } from 'chartshelf';

// points are [timestamp, value], one hour apart
line('#timeline', {
  id: 'traffic',
  title: 'Requests',
  zoom: true,
  markers: false,
  series: [{ name: 'Requests', data: points }],
  x: { format: 'HH:mm' },
  y: { zero: false },
});

area('#overview', {
  height: 132,
  legend: false,
  brush: { target: 'traffic' },
  series: [{ name: 'Requests', data: points }],
});
